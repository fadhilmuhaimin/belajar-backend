# v1 berjalan.
$ docker run -d --name rekeningo-api -p 18082:8080 -e DATABASE_URL=... -e TOKEN_SECRET=... rekeningo-api:v1

# A. Deploy v2 ala Tahap 1: hentikan v1, jalankan v2. Prober: GET /sehat setiap 100 ms.

$ docker rm -f rekeningo-api

$ docker run -d --name rekeningo-api -p 18082:8080 -e DATABASE_URL=... -e TOKEN_SECRET=... rekeningo-api:v2

$ curl http://127.0.0.1:18082/sehat
200 {"status": "ok", "versi": "v2"}  (siap setelah 0.0 detik)

# Ringkasan prober (deploy v2 sukses): 18 request, 2 gagal, tidak bisa dipakai sekitar 0.2 detik.

# B. Deploy v3, tapi konfigurasi deploy lupa TOKEN_SECRET.

$ docker run -d --name rekeningo-api -p 18082:8080 -e DATABASE_URL=... rekeningo-api:v3
