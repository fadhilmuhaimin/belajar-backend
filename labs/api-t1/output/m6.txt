# laptop: adalah folder build di laptop Raka, vps: adalah /opt/rekeningo di VPS. Di lab, keduanya folder sementara.

# --- A. Kode v2: kolom transaksi.jumlah diganti nominal, field JSON ikut berganti ---
#   internal/handler/handler.go  Jumlah *int64 `json:"jumlah"`  ->  Jumlah *int64 `json:"nominal"`
#   internal/handler/handler.go  []string{"ke", "jumlah"}  ->  []string{"ke", "nominal"}
#   internal/service/bayar.go    ErrValidasi{"jumlah",  ->  ErrValidasi{"nominal",
#   internal/repo/transaksi.go   INSERT INTO transaksi (dari, ke, jumlah)  ->  INSERT INTO transaksi (dari, ke, nominal)
#   internal/repo/transaksi.go   SELECT id, dari, ke, jumlah, dibuat  ->  SELECT id, dari, ke, nominal, dibuat
#   internal/repo/transaksi.go   Jumlah int64     `json:"jumlah"`  ->  Jumlah int64     `json:"nominal"`
#   internal/repo/riwayat.go     Jumlah int64     `json:"jumlah"`  ->  Jumlah int64     `json:"nominal"`
#   internal/repo/riwayat.go     a.nama, t.jumlah, t.dibuat  ->  a.nama, t.nominal, t.dibuat
#   internal/repo/riwayat.go     sum(jumlah)  ->  sum(nominal)
$ go build -o laptop:api-t1 ./cmd/api   # di laptop Raka, kode v2
