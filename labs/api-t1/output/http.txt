# --- B. GET membaca, tidak mengubah apa pun ---
$ curl -i -X GET '/akun/419' -H 'Authorization: Bearer <token sesi Budi>'
HTTP/1.1 200 OK
Content-Type: application/json

{"id":419,"jenis":"karyawan","nama":"Budi","saldo":250000}

# --- C. POST membuat satu transaksi: 201 dan header Location ---
$ curl -i -X POST '/transfers' -H 'Authorization: Bearer <token sesi Budi>' -d '{"ke": 418, "jumlah": 25000}'
HTTP/1.1 201 Created
Content-Type: application/json
Location: /transfers/1

{"id":1,"saldo":225000}

# --- D. Alamat di Location bisa dibuka ---
$ curl -i -X GET '/transfers/1' -H 'Authorization: Bearer <token sesi Budi>'
HTTP/1.1 200 OK
Content-Type: application/json

{"id":1,"dari":419,"ke":418,"jumlah":25000,"waktu":"<hari ini>T<jam>+07:00"}

# --- E. Method yang tidak didaftarkan untuk path ini: 405 dan header Allow ---
$ curl -i -X GET '/transfers' -H 'Authorization: Bearer <token sesi Budi>'

Allow: POST
Content-Type: text/plain; charset=utf-8

$ curl -i -X DELETE '/transfers/1' -H 'Authorization: Bearer <token sesi Budi>'
