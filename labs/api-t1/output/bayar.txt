# --- D. Yang diterima PostgreSQL selama pembayaran: satu transaction ---
$ docker compose logs db | grep 'api-t1|'   (hanya statement pembayaran)
begin
UPDATE akun SET saldo = saldo - $1 WHERE id = $2 RETURNING saldo
UPDATE akun SET saldo = saldo + $1 WHERE id = $2
INSERT INTO transaksi (dari, ke, jumlah) VALUES ($1, $2, $3) RETURNING id
commit
