# --- D. Rekonsiliasi akun Budi dari catatan: top-up - keluar + masuk + koreksi ---
$ psql -c "SELECT a.saldo, t.topup, k.keluar, m.masuk, x.koreksi, t.topup - k.keluar + m.masuk + x.koreksi AS dari_catatan FROM akun a, (SELECT coalesce(sum(nominal), 0) AS topup FROM topup WHERE akun_id = 419) t, (SELECT coalesce(sum(jumlah), 0) AS keluar FROM transaksi WHERE dari = 419) k, (SELECT coalesce(sum(jumlah), 0) AS masuk FROM transaksi WHERE ke = 419) m, (SELECT coalesce(sum(jumlah), 0) AS koreksi FROM koreksi WHERE akun_id = 419) x WHERE a.id = 419"
